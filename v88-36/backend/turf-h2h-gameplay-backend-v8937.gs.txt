function turfH2HGameInit(matchId, gameId, slot, username) {
  return turfH2HWithGameLock_(function(){

    if(!st){ st=turfH2HNewGame_(matchId,gameId); }
    if(st.gameId!==String(gameId||'')) return {ok:false,error:'wrong_game'};
    slot=turfH2HSlot_(slot); if(!slot) return {ok:false,error:'bad_slot'};
    st.players[slot].name=turfH2HSafeName_(username,slot);
    st.players[slot].ready=true;
    st.updatedAt=Date.now();
    turfH2HSaveGame_(st);
    return turfH2HPublicGame_(st,slot);
  });
}

function turfH2HGameChooseMove(matchId, slot, move){

    if(!st||!slot)return {ok:false,error:'missing'};

  });
}
function turfH2HGameAnswer(matchId, slot, answerIndex){
  return turfH2HWithGameLock_(function(){
    var st=turfH2HLoadGame_(matchId); slot=turfH2HSlot_(slot);
    if(!st||!slot||!st.pending||st.pending.slot!==slot)return {ok:false,error:'no_question'};

    if(correct){
      if(st.gameId==='ttt') st.board[Number(st.pending.move)]=slot;
      if(st.gameId==='connect4') turfH2HDrop_(st.board,Number(st.pending.move),slot);
    }

    st.pending=null; st.moveNo++;
    var winner=turfH2HWinner_(st);

    else st.turn=(slot==='A'?'B':'A');

  });

function turfH2HCasesAction(matchId, slot, action, value){
 return turfH2HWithGameLock_(function(){
  var st=turfH2HLoadGame_(matchId); slot=turfH2HSlot_(slot);

  if(st.status!=='active'||st.turn!==slot)return {ok:false,error:'not_your_turn'};
  var p=st.cases[slot]; action=String(action||'');
  if(action==='choose'){
    var n=Number(value); if(p.myCase||n<1||n>10||p.opened[n])return {ok:false,error:'bad_case'}; p.myCase=n;
  } else if(action==='open'){
    var c=Number(value); if(!p.myCase||p.offer||p.finalChoice||c<1||c>10||c===p.myCase||p.opened[c])return {ok:false,error:'bad_case'};
    p.opened[c]=true; p.openedThisPhase++;
    var needed=[3,3,2][Math.min(p.stage,2)];
    if(p.openedThisPhase>=needed){p.offer=turfH2HMakeOffer_(st,p);p.openedThisPhase=0;}
  } else if(action==='offer'){
    if(!p.offer)return {ok:false,error:'no_offer'};
    var accept=String(value)==='accept';
    if(accept){p.lockedPlayer={name:p.offer.name,projection:p.offer.projection,source:'BANKER'};p.offer=null;p.complete=true;turfH2HAfterCasesTurn_(st,slot);}
    else {p.offer=null;if(p.stage>=2){p.finalChoice=true;}else{p.stage++;turfH2HAfterCasesTurn_(st,slot);}}
  } else if(action==='final'){
    if(!p.finalChoice)return {ok:false,error:'no_final'};
    var remaining=turfH2HRemainingCase_(p); var chosen=(String(value)==='swap'?remaining:p.myCase);

  } else return {ok:false,error:'bad_action'};
  st.updatedAt=Date.now();turfH2HSaveGame_(st);return turfH2HPublicGame_(st,slot);

function turfH2HRemainingCase_(p){for(var k=1;k<=10;k++)if(k!==p.myCase&&!p.opened[k])return k;return p.myCase;}
